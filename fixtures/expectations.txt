# Hand-authored differential expectation table (independent oracle).
# Naming in kept/dropped sets: "<object-name>:<section-name>", where
# object-name is the .object name declared inside the .asm source.
spec = 1

# Weak provider alone: 2 + 2 = 4.
case weak_only
objects = ws_main.mkobj ws_weak.mkobj
entry = main
expect = ok
return = 4
kept = ws_main.o:.text.main ws_weak.o:.text.provider

# Strong provider overrides weak: weak section is reclaimed: 2 + 40 = 42.
case weak_strong_override
objects = ws_main.mkobj ws_weak.mkobj ws_strong.mkobj
entry = main
expect = ok
return = 42
kept = ws_main.o:.text.main ws_strong.o:.text.provider
dropped = ws_weak.o:.text.provider

# Mutual recursion across two objects: iseven(10) = 1.
case circular_iseven
objects = circ_main.mkobj circ_even.mkobj circ_odd.mkobj
entry = main
expect = ok
return = 1
kept = circ_main.o:.text.main circ_even.o:.text.iseven circ_odd.o:.text.isodd

# Strong undefined symbol in a reachable section is diagnosed.
case undefined_symbol

# Exported symbol is an indirect GC root; keep section is also a root.

# Two strong definitions: deterministic conflict, link never happens.

# Transitive chain main -> a -> b -> c across four objects.

return = 7
kept = chain_main.o:.text.main chain_a.o:.text.a chain_b.o:.text.b chain_c.o:.text.c
